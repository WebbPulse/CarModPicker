        render(<ReportReview />, testScenarios.adminAuthenticated);
        await waitFor(() => expect(apiClient.get).toHaveBeenCalled());

        const listItems = screen.queryAllByRole('listitem');
        expect(rows.length + listItems.length).toBeGreaterThan(0);
      });

      it('approves a report via action button', async () => {
        vi.mocked(apiClient.post).mockResolvedValueOnce({ data: { message: 'approved' } });

        const approveButton = await screen.findByRole('button', { name: /approve|accept/i });
        await user.click(approveButton);
        await waitFor(() =>
          expect(apiClient.post).toHaveBeenCalledWith(
            expect.stringMatching(/\/admin\/reports\/[^/]+\/approve/),
            expect.anything()
          )
        );
      });

      it('rejects a report via action button', async () => {
        vi.mocked(apiClient.post).mockResolvedValueOnce({ data: { message: 'rejected' } });
        const user = userEvent.setup();

        const rejectButton = await screen.findByRole('button', { name: /reject|deny/i });
        await user.click(rejectButton);
        await waitFor(() =>
          expect(apiClient.post).toHaveBeenCalledWith(
            expect.stringMatching(/\/admin\/reports\/[^/]+\/reject/),
            expect.anything()
          )
        );
      });

      it('denies access to non-admin user', () => {
        render(<ReportReview />, testScenarios.authenticated);
        expect(screen.getByText(/access denied|admin only|forbidden/i)).toBeInTheDocument();
      });
    });
