    ```ts
    import { makeAdminUserView, makeUserList } from '../test/mocks/admin/users';

    describe('adminApi — system stats', () => {
      beforeEach(() => vi.clearAllMocks());

      it('getSystemStats hits /admin/stats with GET', async () => {
        const stats = makeSystemStats();
        vi.mocked(apiClient.get).mockResolvedValueOnce({ data: stats });
        const result = await adminApi.getSystemStats();
        expect(apiClient.get).toHaveBeenCalledWith('/admin/stats');
        expect(result.data).toEqual(stats);
      });
    });

    describe('adminApi — user management', () => {
      beforeEach(() => vi.clearAllMocks());

      it('listUsers hits /admin/users with GET', async () => {
        vi.mocked(apiClient.get).mockResolvedValueOnce({ data: makeUserList() });
        await adminApi.listUsers();
        expect(apiClient.get).toHaveBeenCalledWith('/admin/users', expect.anything());
      });

      it('banUser posts to /admin/users/:id/ban', async () => {
        vi.mocked(apiClient.post).mockResolvedValueOnce({ data: { message: 'banned' } });
        await adminApi.banUser('user-123', 'Spam');
        expect(apiClient.post).toHaveBeenCalledWith(
          '/admin/users/user-123/ban',
          expect.objectContaining({ reason: 'Spam' })
        );
      });
    });
    ```
