    ```tsx
    import { describe, it, expect, beforeEach, vi } from 'vitest';
    import userEvent from '@testing-library/user-event';
    import { render, screen, waitFor, testScenarios } from '../../test/utils/test-utils';
    import { apiClient } from '../../api/client';
    import { mockPart } from '../../test/mocks/api';
    import EditPart from './EditPart';

    describe('EditPart page', () => {
      beforeEach(() => {
        vi.clearAllMocks();
        vi.mocked(apiClient.get).mockResolvedValue({ data: mockPart });
      });

      it('renders part data in editable fields', async () => {
        render(<EditPart />, {
          route: `/parts/${mockPart.id}/edit`,
          ...testScenarios.authenticated,
        });
        await waitFor(() => expect(screen.getByDisplayValue(mockPart.name)).toBeInTheDocument());
      });

      it('submits edits and calls apiClient.put/patch with updated body', async () => {
        vi.mocked(apiClient.put).mockResolvedValueOnce({ data: { ...mockPart, name: 'Updated' } });
        vi.mocked(apiClient.patch).mockResolvedValueOnce({ data: { ...mockPart, name: 'Updated' } });
        const user = userEvent.setup();
        render(<EditPart />, {
          route: `/parts/${mockPart.id}/edit`,
          ...testScenarios.authenticated,
        });
        const nameField = await screen.findByDisplayValue(mockPart.name);
        await user.clear(nameField);
        await user.type(nameField, 'Updated');
        await user.click(screen.getByRole('button', { name: /save|update/i }));
        await waitFor(() => {
          const putCalls = vi.mocked(apiClient.put).mock.calls.length;
          const patchCalls = vi.mocked(apiClient.patch).mock.calls.length;
          expect(putCalls + patchCalls).toBeGreaterThan(0);
        });
      });
    });
    ```
