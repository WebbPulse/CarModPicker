    ```tsx
    import { describe, it, expect, beforeEach, vi } from 'vitest';
    import userEvent from '@testing-library/user-event';
    import { render, screen, waitFor, testScenarios } from '../../test/utils/test-utils';
    import { apiClient } from '../../api/client';
    import { makeCurationCandidate, makeCurationQueue } from '../../test/mocks/admin/curation';
    import PartsCuration from './PartsCuration';

    describe('PartsCuration page', () => {
      beforeEach(() => {
        vi.clearAllMocks();
        vi.mocked(apiClient.get).mockResolvedValue({ data: makeCurationQueue() });
      });

      it('renders curation queue for admin', async () => {
        render(<PartsCuration />, testScenarios.adminAuthenticated);
        await waitFor(() => expect(apiClient.get).toHaveBeenCalled());
        // Assert queue visible
        const text = document.body.textContent || '';
        expect(text.length).toBeGreaterThan(0);
      });

      it('approves a curation candidate', async () => {
        vi.mocked(apiClient.post).mockResolvedValueOnce({ data: { message: 'approved' } });
        const user = userEvent.setup();
        render(<PartsCuration />, testScenarios.adminAuthenticated);
        const approveButton = await screen.findByRole('button', { name: /approve|accept/i });
        await user.click(approveButton);
        await waitFor(() =>
          expect(apiClient.post).toHaveBeenCalledWith(
            expect.stringMatching(/\/admin\/(curation|parts)/),
            expect.anything()
          )
        );
      });

      it('merges a candidate into canonical', async () => {
        vi.mocked(apiClient.post).mockResolvedValueOnce({ data: { message: 'merged' } });
        const user = userEvent.setup();
        render(<PartsCuration />, testScenarios.adminAuthenticated);
        const mergeButton = await screen.findByRole('button', { name: /merge/i });
        await user.click(mergeButton);
        await waitFor(() =>
          expect(apiClient.post).toHaveBeenCalledWith(
            expect.stringContaining('/admin/parts/merge'),
            expect.anything()
          )
        );
      });

      it('denies access to non-admin user', () => {
        render(<PartsCuration />, testScenarios.authenticated);
        expect(screen.getByText(/access denied|admin only|forbidden/i)).toBeInTheDocument();
      });
    });
    ```
