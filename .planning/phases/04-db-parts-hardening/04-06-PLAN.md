"""DATA-10 regression: lazy='raise' relationships fail loud on unloaded access,
succeed when caller pre-loads via selectinload."""
from __future__ import annotations

import pytest
from sqlalchemy import select
from sqlalchemy.exc import InvalidRequestError
from sqlalchemy.orm import Session, selectinload

from app.api.models.build_list import BuildList as DBBuildList
from app.api.models.build_log import BuildLog as DBBuildLog, BuildLogPost as DBBuildLogPost
from app.api.models.user import User as DBUser


def test_build_log_post_author_raises_without_selectinload(db_session: Session, test_user: DBUser) -> None:
    # Seed a BuildLogPost.
    bl = DBBuildList(name="lazy_raise_bl", user_id=test_user.id)
    db_session.add(bl)
    db_session.flush()
    blog = DBBuildLog(build_list_id=bl.id, title="t")
    db_session.add(blog)
    db_session.flush()
    post = DBBuildLogPost(build_log_id=blog.id, user_id=test_user.id, content="x")
    db_session.add(post)
    db_session.commit()

    # Fetch without selectinload
    fetched = db_session.scalars(
        select(DBBuildLogPost).where(DBBuildLogPost.id == post.id)
    ).first()
    db_session.expire(fetched)  # force a lazy-load on next access
    with pytest.raises(InvalidRequestError):
        _ = fetched.author


def test_build_log_post_author_works_with_selectinload(db_session: Session, test_user: DBUser) -> None:
    # Seed a BuildLogPost.
    bl = DBBuildList(name="lazy_raise_bl2", user_id=test_user.id)
    db_session.add(bl)
    db_session.flush()
    blog = DBBuildLog(build_list_id=bl.id, title="t")
    db_session.add(blog)
    db_session.flush()
    post = DBBuildLogPost(build_log_id=blog.id, user_id=test_user.id, content="x")
    db_session.add(post)
    db_session.commit()

    fetched = db_session.scalars(
        select(DBBuildLogPost)
        .where(DBBuildLogPost.id == post.id)
        .options(selectinload(DBBuildLogPost.author))
    ).first()
    assert fetched.author is not None  # eager-loaded; no exception


def test_build_list_parts_raises_without_selectinload(db_session: Session, test_user: DBUser) -> None:

### Procedure

1. Start a local Postgres instance:
   ```
   docker-compose -f docker-compose.test.yml up -d postgres-test
   ```
   (or use `backend/docker-compose.yml` if you prefer the dev-default 5432.)

   ```
   cd backend
   ./scripts/test_migration_round_trip.sh <revision_id>
   ```
   The script runs:
   ```
   alembic upgrade <revision_id>
   alembic downgrade -1
   alembic upgrade head
   ```
   and exits non-zero if any step fails.

3. Paste the script's green output into the PR conversation under a

### Exceptions

- **Forward-only data migrations** — downgrade() is annotated with the
  `# SAFE: forward-only data backfill; no reversal needed` convention (Phase
  1 SAFE-04). The round-trip still runs, but the downgrade is expected to be
  a no-op.
- **Already-applied migrations on prod** — historic revisions that pre-date
  this convention are grandfathered (see Phase 1 SAFE-08 repair pattern).

### Rationale

`alembic downgrade -1` followed by `upgrade head` is the smallest invariant
that catches the most common migration-authoring bugs: missing `op.drop_*`
in `downgrade()`, mismatched column types, naming-convention drift, etc.
Automating in CI is a future task; a reviewer-gated doc check is sufficient
for current team size (one developer).
