```python
# backend/app/core/cloudwatch_emf.py skeleton
from __future__ import annotations

import logging
import os

from aws_embedded_metrics import metric_scope
from aws_embedded_metrics.logger.metrics_logger import MetricsLogger

from app.core.config import settings

logger = logging.getLogger(__name__)


def emit_crawler_run_metrics(
    *,
    adapter_name: str,
    run_type: str,  # "live" or "rescrape"
    ingested: int,
    parse_failures: int,
    elapsed_seconds: float,
) -> None:
    """Emit one EMF record per crawler run. No-op outside staging/production.

    Dimensions: AdapterName × Environment × RunType (D-19 — plan 02-05 filters
    RunType=live so rescrape noise doesn't trigger the parse-failure alarm).

    Metrics:
      - Ingested (Count)        — from runner.py `ingested` local
      - ParseFailures (Count)   — from runner.py `skipped_not_product` local (D-22)
      - ElapsedSeconds (Seconds) — monotonic time around the URL loop

    Critical (RESEARCH Landmine 3 / aws-embedded-metrics #109): call this BEFORE
    the `logger.log(summary_level, "Adapter %s done...")` line. The summary line
    acts as the non-EMF flush trigger that prevents awslogs from dropping the
    final EMF line. NEVER reorder.

    Critical (RESEARCH Landmine 4): `AWS_EMF_ENVIRONMENT=Local` MUST be set in
    the runtime env (plan 02-02 terraform adds this to apprunner.tf + ecs.tf).
    Without it the library falls back to a CloudWatch agent sink that doesn't
    exist on ECS Fargate and silently drops metrics.
    """
    if os.environ.get("TESTING") == "true":
        return
    env = (settings.APP_ENVIRONMENT or "").lower()
    if env not in {"staging", "production"}:
        return

    try:
        _emit_scoped(
            adapter_name=adapter_name,
            env=env,
            run_type=run_type,
            ingested=ingested,
            parse_failures=parse_failures,
            elapsed_seconds=elapsed_seconds,
        )
    except Exception as exc:  # email.py analog: log + continue, never crash run_crawler
        logger.error(
            "emit_crawler_run_metrics failed for adapter=%s run_type=%s: %s",
            adapter_name, run_type, exc,
        )


@metric_scope  # auto-flush on function exit — handles sync/async dance (Landmine 5)
def _emit_scoped(
    metrics: MetricsLogger,  # injected by @metric_scope
    *,
    adapter_name: str,
    env: str,
    run_type: str,
    ingested: int,
    parse_failures: int,
    elapsed_seconds: float,
) -> None:
    metrics.set_namespace("CarModPicker/Crawlers")
    metrics.set_dimensions(
        {
            "AdapterName": adapter_name,
            "Environment": env,
            "RunType": run_type,
        }
    )
    metrics.put_metric("Ingested", ingested, "Count")
    metrics.put_metric("ParseFailures", parse_failures, "Count")
    metrics.put_metric("ElapsedSeconds", elapsed_seconds, "Seconds")
```
