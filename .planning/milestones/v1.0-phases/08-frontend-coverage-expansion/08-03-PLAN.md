Canonical test skeleton: PATTERNS.md §7. Every test file follows this shape:
```ts
import { beforeEach, describe, expect, it, vi } from 'vitest';
import { apiClient } from './client';
import { xApi } from './x';
import { mockX } from '../test/mocks/api';

describe('xApi', () => {
  beforeEach(() => vi.clearAllMocks());

  it('someMethod hits /path with VERB', async () => {
    vi.mocked(apiClient.get).mockResolvedValueOnce({ data: mockX });
    const result = await xApi.someMethod('arg');
    expect(apiClient.get).toHaveBeenCalledWith('/exact/path/arg');
    expect(result.data).toEqual(mockX);
  });
});
```
