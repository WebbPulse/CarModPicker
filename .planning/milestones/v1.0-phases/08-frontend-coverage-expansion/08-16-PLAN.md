    import { apiClient } from '../../api/client';
    import { makeUserList, makeAdminUserView } from '../../test/mocks/admin/users';
    import UserManagement from './UserManagement';

    describe('UserManagement page', () => {

          // Assert at least one username from makeUserList is visible

          expect(textContent.length).toBeGreaterThan(0);
        });
        expect(apiClient.get).toHaveBeenCalledWith(expect.stringContaining('/admin/users'), expect.anything());
      });

      it('bans a user via action button', async () => {
        vi.mocked(apiClient.post).mockResolvedValueOnce({ data: { message: 'banned' } });
        const user = userEvent.setup();

        await waitFor(() => screen.queryAllByRole('row').length > 0 || true);
        const banButton = await screen.findByRole('button', { name: /ban|suspend/i });
        await user.click(banButton);
        await waitFor(() =>
          expect(apiClient.post).toHaveBeenCalledWith(
            expect.stringMatching(/\/admin\/users\/[^/]+\/ban/),

          )
        );

      it('denies access to non-admin user', () => {
        render(<UserManagement />, testScenarios.authenticated);
        expect(screen.getByText(/access denied|admin only|forbidden/i)).toBeInTheDocument();
