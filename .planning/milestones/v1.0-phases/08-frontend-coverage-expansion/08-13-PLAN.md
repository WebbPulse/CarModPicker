    import { describe, it, expect, beforeEach, vi } from 'vitest';
    import { render, screen, waitFor, testScenarios } from '../../test/utils/test-utils';

    import { mockBuildList, mockUser } from '../../test/mocks/api';
    import BuildListsCatalog from './BuildListsCatalog';

    describe('BuildListsCatalog page', () => {
      beforeEach(() => {
        vi.clearAllMocks();
        vi.mocked(apiClient.get).mockResolvedValue({ data: [mockBuildList] });
      });

      it('renders build list catalog', async () => {
        render(<BuildListsCatalog />, {
          route: '/build-lists',
          ...testScenarios.unauthenticated,
        });
        await waitFor(() => expect(screen.getByText(mockBuildList.name)).toBeInTheDocument());
      });

      it('shows empty state when no build lists', async () => {
        vi.mocked(apiClient.get).mockResolvedValue({ data: [] });
        render(<BuildListsCatalog />, {
          route: '/build-lists',
          ...testScenarios.unauthenticated,
        });
        await waitFor(() => expect(screen.getByText(/no build lists|empty/i)).toBeInTheDocument());
      });

      it('forwards filter params to API', async () => {
        render(<BuildListsCatalog />, {
          route: `/build-lists?car_id=${mockBuildList.car_id}`,
          ...testScenarios.unauthenticated,
        });
        await waitFor(() => expect(apiClient.get).toHaveBeenCalled());
        const calls = vi.mocked(apiClient.get).mock.calls;
        const urlOrOptsHasFilter = calls.some(([url, opts]) => {
          return url.includes(mockBuildList.car_id) ||
            (opts?.params && JSON.stringify(opts.params).includes(mockBuildList.car_id));
        });
        expect(urlOrOptsHasFilter).toBe(true);
      });
    });
    ```

    import { describe, it, expect, beforeEach, vi } from 'vitest';
    import userEvent from '@testing-library/user-event';
    import { render, screen, waitFor, testScenarios } from '../../test/utils/test-utils';
    import { apiClient } from '../../api/client';
    import { mockBuildList, mockUser } from '../../test/mocks/api';
    import ViewBuildLog from './ViewBuildLog';

    describe('ViewBuildLog page', () => {
      beforeEach(() => {
        vi.clearAllMocks();
        vi.mocked(apiClient.get).mockImplementation((url: string) => {
          if (url.includes('/build-logs')) return Promise.resolve({ data: { posts: [], total: 0 } });
          if (url.includes('/build-lists/')) return Promise.resolve({ data: mockBuildList });
          return Promise.resolve({ data: null });
        });
      });

      it('renders build log heading and compose form for authenticated user', async () => {
        render(<ViewBuildLog />, {
          route: `/build-logs/${mockBuildList.id}`,
          ...testScenarios.authenticated,
        });
        await waitFor(() => expect(screen.getByText(mockBuildList.name)).toBeInTheDocument());
        // assert compose form elements (textarea, submit button)
        expect(screen.getByRole('textbox')).toBeInTheDocument();
      });

      it('submits a post to the build log', async () => {
        vi.mocked(apiClient.post).mockResolvedValueOnce({
          data: { id: 'post-1', body: 'Test post', author: mockUser, created_at: '2026-01-01' },
        });
        const user = userEvent.setup();
        render(<ViewBuildLog />, {
          route: `/build-logs/${mockBuildList.id}`,
          ...testScenarios.authenticated,
        });
        await waitFor(() => screen.getByRole('textbox'));
        await user.type(screen.getByRole('textbox'), 'Test post');
        await user.click(screen.getByRole('button', { name: /post|submit/i }));
        await waitFor(() => expect(apiClient.post).toHaveBeenCalled());
      });
    });
    ```
