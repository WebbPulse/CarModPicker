    op.execute(sa.text("""
        INSERT INTO build_logs (id, build_list_id, title, created_at, updated_at)
        SELECT gen_random_uuid(), bl.id, 'Build Log: ' || bl.name, NOW(), NOW()
        FROM build_lists bl

from dataclasses import dataclass, field

_SELECT_PATTERN = re.compile(r"^\s*SELECT\b", re.IGNORECASE)


@dataclass
class QueryCounter:
    count: int = 0
    statements: list[str] = field(default_factory=list)

### Shared Pattern B: `with_for_update()` row-lock (DATA-03)

### Shared Pattern C: selectinload eager-load (DATA-01)

**Source:** RESEARCH.md §1.5 Pattern 2.

**Apply to:** `backend/app/api/endpoints/build_logs.py` posts fetch (DATA-01). Future use in any new endpoint that reads a paginated collection with per-row relationship access.

**Shape:**
```python
from sqlalchemy.orm import selectinload

stmt = (
    select(DBBuildLogPost)

BACKEND_APP = Path(__file__).resolve().parent.parent / "app"
PATTERN = re.compile(r"...")

def test_no_X_in_app() -> None:
    offenders: list[tuple[str, int]] = []
    for pyfile in BACKEND_APP.rglob("*.py"):
        for lineno, line in enumerate(pyfile.read_text(encoding="utf-8").splitlines(), 1):
            if line.lstrip().startswith("#"):
                continue
            if PATTERN.search(line):
                offenders.append((str(pyfile.relative_to(BACKEND_APP)), lineno))
    assert not offenders, f"Found {len(offenders)} offenders: {offenders!r}"
```
